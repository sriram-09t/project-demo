Dijkstra’s Algorithm for Shortest Path

// /* write a C progrram Implement Dijkstra’s algorithm to compute the shortest path through a network.
// Enter number of vertices in the graph: 4
// Enter the adjacency matrix (0 if no edge):
// 0 1 2 3
// 1 0 2 0
// 2 2 0 3
// 3 0 3 0
// Enter the source vertex (0 - 3): 0
// Vertex 	 Distance from Source
// 0 		 0
// 1 		 1
// 2 		 2
// 3 		 3
// */

#include <stdio.h>
#include <limits.h>

#define MAX 100

int main() {
    int n;
    int graph[MAX][MAX];
    int distance[MAX];
    int visited[MAX];
    int source;

    printf("Enter number of vertices in the graph: ");
    scanf("%d", &n);

    printf("Enter the adjacency matrix (0 if no edge):\n");

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            scanf("%d", &graph[i][j]);
        }
    }

    printf("Enter the source vertex (0 - %d): ", n - 1);
    scanf("%d", &source);

    // Initialize distances and visited array
    for (int i = 0; i < n; i++) {
        distance[i] = INT_MAX;
        visited[i] = 0;
    }

    distance[source] = 0;

    // Dijkstra's algorithm
    for (int count = 0; count < n - 1; count++) {
        int min = INT_MAX;
        int u = -1;

        // Find the unvisited vertex with minimum distance
        for (int i = 0; i < n; i++) {
            if (!visited[i] && distance[i] < min) {
                min = distance[i];
                u = i;
            }
        }

        if (u == -1)
            break;

        visited[u] = 1;

        // Update distances of adjacent vertices
        for (int v = 0; v < n; v++) {
            if (!visited[v] &&
                graph[u][v] != 0 &&
                distance[u] != INT_MAX &&
                distance[u] + graph[u][v] < distance[v]) {

                distance[v] = distance[u] + graph[u][v];
            }
        }
    }

    printf("\nVertex \t Distance from Source\n");

    for (int i = 0; i < n; i++) {
        if (distance[i] == INT_MAX)
            printf("%d \t INF\n", i);
        else
            printf("%d \t\t %d\n", i, distance[i]);
    }

    return 0;
}

Distance Vector Routing

/* write a C program that implements the Distance Vector Routing Algorithm to compute the routing table for each node in a network.(using the Bellman-Ford approach)
Enter the number of nodes: 4
Enter the cost matrix (use 'X' if no direct link):
0 1 2 x
1 0 3 4
2 3 0 x
x 4 x 0

Final Routing Tables:

Router 1:
Destination	Cost	Next Hop
1		0	1
2		1	2
3		2	3
4		5	2

Router 2:
Destination	Cost	Next Hop
1		1	1
2		0	2
3		3	3
4		4	4

Router 3:
Destination	Cost	Next Hop
1		2	1
2		3	2
3		0	3
4		7	1

Router 4:
Destination	Cost	Next Hop
1		5	2
2		4	2
3		7	2
4		0	4




*/

#include <stdio.h>
#include <string.h>

#define MAX 100
#define INF 9999

int main() {
    int n;
    int cost[MAX][MAX];
    int distance[MAX][MAX];
    int nextHop[MAX][MAX];

    printf("Enter the number of nodes: ");
    scanf("%d", &n);

    printf("Enter the cost matrix (use 'X' if no direct link):\n");

    // Read cost matrix
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            char input[20];
            scanf("%s", input);

            if (input[0] == 'x' || input[0] == 'X') {
                cost[i][j] = INF;
            } else {
                cost[i][j] = atoi(input);
            }

            if (i == j)
                cost[i][j] = 0;
        }
    }

    // Initialize routing tables
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            distance[i][j] = cost[i][j];

            if (i == j) {
                nextHop[i][j] = i;
            } else if (cost[i][j] != INF) {
                nextHop[i][j] = j;
            } else {
                nextHop[i][j] = -1;
            }
        }
    }

    /*
       Bellman-Ford / Distance Vector updates.
       Keep relaxing routes until no changes occur.
    */
    int changed;

    do {
        changed = 0;

        for (int i = 0; i < n; i++) {
            for (int via = 0; via < n; via++) {

                if (cost[i][via] == INF || i == via)
                    continue;

                for (int dest = 0; dest < n; dest++) {

                    if (distance[via][dest] == INF)
                        continue;

                    int newCost = cost[i][via] + distance[via][dest];

                    if (newCost < distance[i][dest]) {
                        distance[i][dest] = newCost;

                        // First hop from router i is via
                        nextHop[i][dest] = via;

                        changed = 1;
                    }
                }
            }
        }

    } while (changed);

    // Display final routing tables
    printf("\nFinal Routing Tables:\n");

    for (int i = 0; i < n; i++) {

        printf("\nRouter %d:\n", i + 1);
        printf("Destination\tCost\tNext Hop\n");

        for (int j = 0; j < n; j++) {

            printf("%d\t\t", j + 1);

            if (distance[i][j] == INF) {
                printf("INF\t");
            } else {
                printf("%d\t", distance[i][j]);
            }

            if (nextHop[i][j] == -1) {
                printf("-\n");
            } else {
                printf("%d\n", nextHop[i][j] + 1);
            }
        }
    }

    return 0;
}

Subnetting

/* To understand the concept of IP subnetting, calculate subnet mask, network and broadcast addresses, and implement a Java program that performs subnetting calculations for different CIDR values. 
input1 :
Enter IP address (e.g., 192.168.10.5): 20.20.20.1
Enter CIDR value (e.g., 26): 26
output1 :
IP SUBNETTING CALCULATION
Input IP Address: 20.20.20.1/26
Subnet Mask: 255.255.255.192
Network Address: 20.20.20.0
Broadcast Address: 20.20.20.63
Usable Host Range: 20.20.20.1 - 20.20.20.62
Number of Subnets: 4
Number of Hosts per Subnet: 62
Subnet Ranges:
Subnet 1: 20.20.20.0 - 20.20.20.63
Subnet 2: 20.20.20.64 - 20.20.20.127
Subnet 3: 20.20.20.128 - 20.20.20.191
Subnet 4: 20.20.20.192 - 20.20.20.255
*/

#include <stdio.h>
#include <string.h>

unsigned int ipToInt(int a, int b, int c, int d) {
    return ((unsigned int)a << 24) |
           ((unsigned int)b << 16) |
           ((unsigned int)c << 8) |
           (unsigned int)d;
}

void intToIp(unsigned int ip) {
    printf("%u.%u.%u.%u",
           (ip >> 24) & 255,
           (ip >> 16) & 255,
           (ip >> 8) & 255,
           ip & 255);
}

int main() {
    int a, b, c, d;
    int cidr;

    printf("Enter IP address (e.g., 192.168.10.5): ");
    scanf("%d.%d.%d.%d", &a, &b, &c, &d);

    printf("Enter CIDR value (e.g., 26): ");
    scanf("%d", &cidr);

    unsigned int ip = ipToInt(a, b, c, d);

    // Create subnet mask
    unsigned int mask;

    if (cidr == 0)
        mask = 0;
    else
        mask = 0xFFFFFFFF << (32 - cidr);

    // Network address
    unsigned int network = ip & mask;

    // Broadcast address
    unsigned int broadcast = network | (~mask);

    // Total addresses per subnet
    unsigned int totalAddresses = 1U << (32 - cidr);

    // Usable hosts
    unsigned int hostsPerSubnet;

    if (cidr < 31)
        hostsPerSubnet = totalAddresses - 2;
    else
        hostsPerSubnet = totalAddresses;

    // Number of subnets assuming original network is /24
    int numberOfSubnets;

    if (cidr >= 24)
        numberOfSubnets = 1 << (cidr - 24);
    else
        numberOfSubnets = 1;

    printf("\nIP SUBNETTING CALCULATION\n");

    printf("Input IP Address: %d.%d.%d.%d/%d\n",
           a, b, c, d, cidr);

    printf("Subnet Mask: ");
    intToIp(mask);
    printf("\n");

    printf("Network Address: ");
    intToIp(network);
    printf("\n");

    printf("Broadcast Address: ");
    intToIp(broadcast);
    printf("\n");

    printf("Usable Host Range: ");
    intToIp(network + 1);
    printf(" - ");
    intToIp(broadcast - 1);
    printf("\n");

    printf("Number of Subnets: %d\n", numberOfSubnets);
    printf("Number of Hosts per Subnet: %u\n", hostsPerSubnet);

    printf("Subnet Ranges:\n");

    // Base /24 network
    unsigned int baseNetwork = ip & 0xFFFFFF00;

    unsigned int subnetSize = totalAddresses;

    for (int i = 0; i < numberOfSubnets; i++) {

        unsigned int subnetStart =
            baseNetwork + (i * subnetSize);

        unsigned int subnetEnd =
            subnetStart + subnetSize - 1;

        printf("Subnet %d: ", i + 1);

        intToIp(subnetStart);

        printf(" - ");

        intToIp(subnetEnd);

        printf("\n");
    }

    return 0;
}
